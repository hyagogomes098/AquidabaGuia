    <!-- SCRIPT DE FILTRO -->
    <script>
        function filtrar(categoria, elemento) {
            var botoes = document.querySelectorAll('.filtro-btn');
            for (var i = 0; i < botoes.length; i++) {
                botoes[i].classList.remove('ativo');
            }
            elemento.classList.add('ativo');

            var cartoes = document.querySelectorAll('.cartao');
            for (var j = 0; j < cartoes.length; j++) {
                var catCartao = cartoes[j].getAttribute('data-categoria');
                if (categoria === 'todos' || catCartao === categoria) {
                    cartoes[j].classList.remove('escondido');
                } else {
                    cartoes[j].classList.add('escondido');
                }
            }
        }
    </script>
